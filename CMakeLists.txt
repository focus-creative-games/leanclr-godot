cmake_minimum_required(VERSION 3.22)
project(leanclr_godot LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LEANCLR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/extern/leanclr" CACHE PATH "LeanCLR coreclr-minimal checkout")
if(NOT EXISTS "${LEANCLR_SOURCE_DIR}/src/runtime/CMakeLists.txt")
    message(FATAL_ERROR "LeanCLR sources are missing. Run git submodule update --init --recursive, or set LEANCLR_SOURCE_DIR to another checkout.")
endif()
set(GODOTCPP_TARGET template_release CACHE STRING "")
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_THREADS OFF CACHE BOOL "")
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "")
set(LEANCLR_BCL_ADAPTER CORECLR_NET10 CACHE STRING "" FORCE)
if(EMSCRIPTEN)
    set(LEANCLR_HOST PORTABLE CACHE STRING "" FORCE)
    set(LEANCLR_FORCE_PORTABLE ON CACHE BOOL "" FORCE)
else()
    set(LEANCLR_HOST NATIVE CACHE STRING "" FORCE)
    set(LEANCLR_FORCE_PORTABLE OFF CACHE BOOL "" FORCE)
endif()
set(LEANCLR_GC_DEBUG OFF CACHE BOOL "")
set(LEANCLR_ENABLE_CPP_EXCEPTIONS OFF CACHE BOOL "")
add_subdirectory(extern/godot-cpp)
add_subdirectory("${LEANCLR_SOURCE_DIR}/src/runtime" leanclr-runtime)
# This pinned runtime defines a C symbol with the Win32 API's name. Keep the
# runtime-local implementation distinct when linking with godot-cpp's Kernel32 imports.
if(WIN32)
    set_property(SOURCE "${LEANCLR_SOURCE_DIR}/src/runtime/bcl/coreclr-net10/pinvokes/coreclr_qcall.cpp"
        TARGET_DIRECTORY leanclr-bcl-coreclr-net10-objects APPEND PROPERTY COMPILE_DEFINITIONS "GetCPInfo=leanclr_net10_GetCPInfo")
endif()
add_library(leanclr_godot SHARED native/leanclr_host.cpp native/register_types.cpp)
target_link_libraries(leanclr_godot PRIVATE godot-cpp leanclr)
target_include_directories(leanclr_godot PRIVATE "${LEANCLR_SOURCE_DIR}/src/runtime/public")
set_target_properties(leanclr_godot PROPERTIES
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/demo/addons/leanclr/bin"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/demo/addons/leanclr/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(leanclr_godot PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/demo/addons/leanclr/bin")
endforeach()
if(EMSCRIPTEN)
    foreach(layer leanclr-core-objects leanclr-host-objects leanclr-bcl-coreclr-net10-objects)
        target_compile_options(${layer} PRIVATE -sSIDE_MODULE=1 -sSUPPORT_LONGJMP=wasm)
    endforeach()
    target_compile_options(leanclr_godot PRIVATE -fno-exceptions)
    target_link_options(leanclr_godot PRIVATE -sSIDE_MODULE=1 -sWASM_BIGINT -sSUPPORT_LONGJMP=wasm)
    set_target_properties(leanclr_godot PROPERTIES PREFIX "lib" SUFFIX ".wasm" OUTPUT_NAME "leanclr_godot.web")
elseif(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_target_properties(leanclr_godot PROPERTIES OUTPUT_NAME "leanclr_godot.windows.x86_64")
else()
    message(FATAL_ERROR "This minimal demo currently targets Windows x64 and WebAssembly32")
endif()
