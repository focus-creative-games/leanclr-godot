cmake_minimum_required(VERSION 3.22)
project(leanclr_godot_m1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LEANCLR_SOURCE_DIR "${ROOT}/extern/leanclr" CACHE PATH "LeanCLR sources")
set(LEANCLR_BCL_ADAPTER CORECLR_NET10 CACHE STRING "" FORCE)
if(EMSCRIPTEN)
    set(LEANCLR_HOST PORTABLE CACHE STRING "" FORCE)
    set(LEANCLR_FORCE_PORTABLE ON CACHE BOOL "" FORCE)
else()
    set(LEANCLR_HOST NATIVE CACHE STRING "" FORCE)
endif()
set(LEANCLR_GC_DEBUG OFF CACHE BOOL "")
set(LEANCLR_ENABLE_CPP_EXCEPTIONS OFF CACHE BOOL "")
add_subdirectory("${LEANCLR_SOURCE_DIR}/src/runtime" runtime)
if(ANDROID)
    target_compile_definitions(leanclr-bcl-coreclr-net10-objects PRIVATE HAVE_UNISTD_H=1)
endif()
set_property(SOURCE "${LEANCLR_SOURCE_DIR}/src/runtime/bcl/coreclr-net10/pinvokes/coreclr_qcall.cpp"
    TARGET_DIRECTORY leanclr-bcl-coreclr-net10-objects APPEND PROPERTY COMPILE_DEFINITIONS "GetCPInfo=leanclr_net10_GetCPInfo")
add_library(leanclr-godot-calli STATIC "${ROOT}/build/m1/calli/calli_bridges.cpp" "${ROOT}/build/m1/calli/modules.cpp")
set(LEANCLR_GODOT_AOT_DIR "" CACHE PATH "Optional LeanAOT generated gameplay sources")
if(LEANCLR_GODOT_AOT_DIR)
    file(GLOB AOT_SOURCES CONFIGURE_DEPENDS "${LEANCLR_GODOT_AOT_DIR}/*.cpp")
    list(FILTER AOT_SOURCES EXCLUDE REGEX "/calli_bridges\\.cpp$")
    target_sources(leanclr-godot-calli PRIVATE ${AOT_SOURCES})
    target_compile_definitions(leanclr-godot-calli PRIVATE LEANCLR_GODOT_GAMEPLAY_AOT)
endif()
target_link_libraries(leanclr-godot-calli PRIVATE leanclr)
if(MSVC)
    target_compile_options(leanclr-godot-calli PRIVATE /fp:precise)
else()
    target_compile_options(leanclr-godot-calli PRIVATE -ffp-contract=off)
endif()
option(LEANCLR_GODOT_ABI_PROBES "Build the public host ABI regression executable" OFF)
if(LEANCLR_GODOT_ABI_PROBES)
    add_executable(m0-public-abi "${ROOT}/tests/m0/public_abi.cpp")
    target_include_directories(m0-public-abi PRIVATE "${LEANCLR_SOURCE_DIR}/src/runtime/public")
    target_link_libraries(m0-public-abi PRIVATE leanclr)
endif()
